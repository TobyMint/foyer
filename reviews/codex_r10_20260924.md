import csv, os, sys


def load(p):
    return {r["run"]: r for r in csv.DictReader(l for l in open(p) if not l.startswith("#"))}


def trace_ceiling(path):
    """trace 允许的复用上界：所有步 prefix_len 之和，以及 prompt 总量。"""
    rows = list(csv.DictReader(open(path)))

            "ceiling_frac": reuse / prompt,

"""兑现率的逐段斜率 —— paper/data/slope.csv（论文 §VI 的头条数字来源）。

§VI 的核心对比是「两条 trace 的坍塌形状不同」：
  realhr    -31.3  -36.6  -28.4        基本恒定 —— R̄ 1.41 时还没到悬崖
  主 trace  -10.8  -41.0  -157.4  -126.7    R̄ 1.53 后斜率爆炸
这五个数是论文里第一次出现"斜率"这个口径，此前是当场算的、不在任何 CSV 里。

斜率定义：相邻两点的兑现率之差 / 运行数之差，单位是「百分点每单位 R̄」。

用法: python3 gen_slope.py <reuse.csv> <reuse_x.csv> <out.csv>
"""
import csv, sys


def load(p):
    return [r for r in csv.DictReader(l for l in open(p) if not l.startswith("#"))]


def main():
    src = [(("main", r["run"]), float(r["Rbar_busy"]), float(r["realized_frac"]))
           for r in load(sys.argv[1]) if r["family"] == "device_only"]
    # 只取【静态点】——控制器臂不参与前沿/序列，混进来会把两段连成一段
    # （第一版没滤，把 realhr_foyer 也算进去，于是 realhr 变成 3 段而不是 4 段）
    src += [(("xtrace", r["run"]), float(r["Rbar_busy"]), float(r["realized_frac"]))
            for r in load(sys.argv[2]) if "_cap" in r["run"]]
    by = {}
    for (tr, run), r, f in src:
        by.setdefault(tr, []).append((r, f, run))
    rows = []
    for tr, pts in by.items():
        pts.sort()
        for (r0, f0, n0), (r1, f1, n1) in zip(pts, pts[1:]):
            rows.append({
                "trace": tr, "from_run": n0, "to_run": n1,
                "r0": "%.3f" % r0, "r1": "%.3f" % r1,
                "realized0_pct": "%.1f" % (f0 * 100), "realized1_pct": "%.1f" % (f1 * 100),
                "slope_pp_per_unit": "%.1f" % ((f1 - f0) * 100 / (r1 - r0)),
            })
    # 全范围斜率单独出两行：正文用它与另一条 trace 比"整体陡不陡"。
    # ⚠ 必须全范围对全范围 —— 曾经拿 realhr 的【三点】斜率(-34.5)去比主 trace 的
    # 全范围(-66.8)，区间不可比，而且 realhr 自己的全范围其实是 -32.3。
    for tr, pts in by.items():
        pts.sort()
        r0, f0, n0 = pts[0]; r1, f1, n1 = pts[-1]
        rows.append({"trace": tr, "from_run": n0, "to_run": n1,
                     "r0": "%.3f" % r0, "r1": "%.3f" % r1,
                     "realized0_pct": "%.1f" % (f0 * 100),
                     "realized1_pct": "%.1f" % (f1 * 100),
                     "slope_pp_per_unit": "%.2f" % ((f1 - f0) * 100 / (r1 - r0))})
    rows.sort(key=lambda x: (x["trace"], float(x["r0"]), float(x["r1"])))
    header = "\n".join([
        "# 兑现率的逐段斜率 —— 由 scripts/lab/gen_slope.py 生成，勿手改",
        "# slope = Δrealized / ΔR̄，单位「百分点每单位忙时在跑数」",
        "# 用途：§VI 说两条 trace 的【形状】不同 —— 主 trace 是阈值（悬崖），",
        "#   xtrace 在同一段跨度内斜率基本恒定、还没到悬崖。",
        "# 每组的【最后一行】是该 trace 的全范围斜率（正文比整体陡不陡必须用这两行，",
        "# 不能拿一条的三点斜率去比另一条的全范围 —— 2026-09-24 犯过）。",
        "trace,from_run,to_run,r0,r1,realized0_pct,realized1_pct,slope_pp_per_unit"])
    fields = ["trace", "from_run", "to_run", "r0", "r1", "realized0_pct",
              "realized1_pct", "slope_pp_per_unit"]
    body = header + "\n" + "\n".join(",".join(str(x[k]) for k in fields) for x in rows) + "\n"
    assert isinstance(body, str)
    with open(sys.argv[3], "w") as fh:
        fh.write(body)
    print("写入 %s：%d 行" % (sys.argv[3], len(rows)))
    for x in rows:
        print("  %-7s R̄ %s->%s  兑现 %5s->%5s  斜率 %s" % (
            x["trace"], x["r0"], x["r1"], x["realized0_pct"], x["realized1_pct"],
            x["slope_pp_per_unit"]))


if __name__ == "__main__":
    main()

def _envelope(keys, runs):
    """候选锚点的上左包络：按墙钟升序扫，SLO 破纪录的才留在前沿上。

    本函数仍用来决定【谁在前沿上】（表里标 anchor 用），但 Δ 的插值不自己算 ——
    见 main() 里对 frontier.py 的调用。
    """
    out, best = [], -1
    for k in sorted(keys, key=lambda k: float(runs[k]["wall_min"])):
        s = float(runs[k]["slo_pct"])
        if s > best:
            out.append((float(runs[k]["wall_min"]), s, k)); best = s
    return out

    """把 frontier.py 作为唯一的前沿定义与插值来源 import 进来。

# 采样周期。metrics.csv 每 5 秒一行。
SAMPLE_S = 5.0
# 采样跨度允许比 run_summary 的墙钟长多少（秒）。采样器比第一个请求早启动、
# 比最后一个请求晚停下，这个偏移在全部 run 上是恒定的 ~6 分钟。
SPAN_SLACK_S = (0.0, 12 * 60)


def read_metrics(path):
    """返回 [(ts, num_running_reqs_or_None)]。"""
    rows = []
    with open(path) as f:
        hdr = f.readline().strip().split(",")
        if "sglang:num_running_reqs" not in hdr:
            raise RuntimeError("%s 没有 sglang:num_running_reqs 列" % path)
        i_ts = hdr.index("ts")
        i_rn = hdr.index("sglang:num_running_reqs")
        for line in f:
            c = line.rstrip("\n").split(",")
            if len(c) <= max(i_ts, i_rn):
                continue
            try:
                ts = float(c[i_ts])
            except ValueError:
                continue
            try:
                rn = float(c[i_rn])
            except ValueError:
                rn = None
            rows.append((ts, rn))
    if len(rows) < 10:
        raise RuntimeError("%s 只有 %d 个采样点" % (path, len(rows)))
    return rows


def decompose(rows):
    """把相邻样本之间的间隔按【前一个】样本的状态归类。

    这是论文 C1 的定义。返回分钟数；残差必须为 0。
    """
    span = rows[-1][0] - rows[0][0]
    busy = idle = other = 0.0
    for (t0, r0), (t1, _) in zip(rows, rows[1:]):
        dt = t1 - t0
        if dt <= 0 or dt > SAMPLE_S * 3:      # 采样断档：既不算 busy 也不算 idle
            other += dt
            continue
        if r0 is None:
            other += dt
        elif r0 == 0:
            idle += dt
        else:
            busy += dt
    resid = span - busy - idle - other
    assert abs(resid) < 1e-6, "分解残差 %.6f s != 0" % resid
    assert other == 0.0, "有 %.1f s 采样断档，该 run 的分解不可信" % other
    return span / 60, busy / 60, idle / 60, 100 * idle / span


def read_steps(path):
    out = []
    with open(path) as f:
        for line in f:
            line = line.strip()
            if line:
                try:
                    out.append(json.loads(line))
                except json.JSONDecodeError:
                    pass
    return out


def step_perreq_ms(steps):
    """方法二：逐请求 decode 间隔的中位。只取 output_len >= 32 的步。"""
    vals = []
    for s in steps:
        n = s.get("output_len_actual") or 0
        tot = s.get("total_duration_ms")
        ft = s.get("first_token_ms")
        if n >= 32 and tot is not None and ft is not None and n > 1:
            vals.append((tot - ft) / (n - 1))
    return statistics.median(vals) if vals else None


def main():
    root = sys.argv[1]
    out_path = sys.argv[2]
    runs = sys.argv[3:]
    if not runs:
        runs = sorted(os.path.basename(os.path.dirname(p))
                      for p in glob.glob(os.path.join(root, "*", "metrics.csv")))

    rows, bad = [], []
    for r in runs:
        d = os.path.join(root, r)
        try:
            rows_m = read_metrics(os.path.join(d, "metrics.csv"))
            span_min, busy_min, idle_min, idle_pct = decompose(rows_m)
            steps = read_steps(os.path.join(d, "steps.jsonl"))
            tokens = sum(s.get("output_len_actual") or 0 for s in steps)

            # 墙钟的口径必须和 run_summary.py 逐字一致，否则图 2 的柱子长度
            # 和图 1 的横坐标就是两个数。见 run_summary.py:91：
            #   墙钟 = 最后一个完成 − 最早一个提交
            # 不能用 st[0]：steps.jsonl 是按完成时间排的。
            subs = [s["submit_timestamp"] for s in steps if s.get("submit_timestamp")]
            comps = [s["complete_timestamp"] for s in steps if s.get("complete_timestamp")]
            if not subs or not comps:
                bad.append((r, "steps.jsonl 缺 submit/complete 时间戳"))
                continue
            wall_min = (max(comps) - min(subs)) / 60.0

            if not (SPAN_SLACK_S[0] <= (span_min - wall_min) * 60 <= SPAN_SLACK_S[1]):
                bad.append((r, "采样跨度 %.1f 分与墙钟 %.1f 分差 %.1f 分，超出容许的 0-12 分"
                            % (span_min, wall_min, span_min - wall_min)))
                continue

            running_all = statistics.mean(v for _, v in rows_m if v is not None)
            # τ 的分母必须是【busy 期间的条件均值】，不是全时段均值。
            #
            # 式子 W_busy ≈ G/R̄ · τ 里，R̄ 是「每一步在跑几个请求」；而全时段
            # 采样均值把空转的那些 0 也算进去了，等于 R̄_busy × busy 占比。
            # 拿它乘 busy 时间，得到的 τ 会系统性地【低】一个 busy 占比：
            # cap1 空转 36.3%，于是 18.5ms 而不是真实的 ~28.9ms。
            # 这个错误曾经让「两个测量相差 12% 以内」这句话成立 —— 其实是
            # 两个错误互相抵消。
            busy_frac = busy_min / span_min if span_min else 1.0
            running = running_all / busy_frac if busy_frac > 0 else running_all
            step_back = (busy_min * 60 * running / tokens * 1000) if tokens else None
            rows.append({
                "run": r.replace("pois200_", ""),
                "wall_min": round(wall_min, 1),
                "span_min": round(span_min, 1),
                "idle_min": round(idle_min, 1),
                "busy_min": round(busy_min, 1),
                "idle_pct": round(idle_pct, 1),
                "running": round(running, 2), "running_all": round(running_all, 2),
                "step_back_ms": round(step_back, 1) if step_back else "",
                "step_perreq_ms": (round(step_perreq_ms(steps), 1)
                                   if step_perreq_ms(steps) else ""),
                # τ₁/τ₂ 超出比例。两个量都非空才算，否则留空。
                "excess_pct": (round(100 * (step_back / step_perreq_ms(steps) - 1), 1)
                               if (step_back and step_perreq_ms(steps)) else ""),
            })
        except Exception as e:
            bad.append((r, str(e)))

    if bad:
        print("!! 以下 run 未通过校验，未写入 CSV：")
        for r, why in bad:
            print("   %-24s %s" % (r, why))

    with open(out_path, "w", newline="") as f:
        f.write("# 墙钟的两段分解 + 步时间 —— 由 scripts/lab/gen_decomp.py 生成，勿手改\n")
        f.write("# busy/idle: 相邻采样按前一时刻 sglang:num_running_reqs 是否为 0 归类；busy+idle==span（有断言）\n")
        f.write("# span_min: metrics.csv 的采样跨度；wall_min: run_summary 的墙钟。两者差 ~6 分，是采样边界\n")
        f.write("# step_back: busy分钟 x 在跑均值 / 产出token（导出量；用 busy 不用墙钟，见下）; step_perreq: 逐请求 decode 间隔中位（独立测量）\n")
        w = csv.DictWriter(f, fieldnames=HEADER)
        w.writeheader()
        w.writerows(rows)
    print("写入 %s：%d 行" % (out_path, len(rows)))


if __name__ == "__main__":
    main()

"""round-1+ 命中率序列 —— paper/data/hitrate.csv。

为什么要有：§related 引用「无 HiCache 时 80.8/78.5/70.6/56.0/32.8%，有 HiCache 时
80.7/79.6/78.7%，控制器 80.6%」以及「计划值 80.9%」。这一串数是从 steps.jsonl 的
server_prefix_hit_rate 逐步算的，此前不在任何 CSV 里（审计报了 8 处"找不到源头"）。

⚠ 口径（正文必须与表头一致，混了会被审稿人抓）：
  这是【观测】的 server 命中率，round>=1 的【逐步非加权均值】。
  它与 gen_reuse.py 的 reuse_frac 是同一现象的不同归一化：
    - 本表：只取 round>=1，逐步非加权
    - reuse_frac：全部步，按 prompt token 加权
  两者不可互相换算，也不能混引。计划的 round-1+ 值恒为 80.9%，不随 cap 变。

用法: python3 gen_hitrate.py <results_dir> <out.csv> <run> [<run> ...]
"""
import csv, json, os, statistics, sys


def round1_mean(path):
    st = [json.loads(l) for l in open(path) if l.strip()]
    r1 = [s for s in st if s.get("round_idx", 0) >= 1]
    if not r1:
        return None, None, 0
    obs = statistics.mean(s.get("server_prefix_hit_rate") or 0 for s in r1)
    plan = statistics.mean(s.get("planned_prefix_hit_rate") or 0 for s in r1)
    return obs, plan, len(r1)


def main():
    root, out_p = sys.argv[1], sys.argv[2]
    runs = sys.argv[3:]
    rows = []
    for r in runs:
        p = os.path.join(root, r, "steps.jsonl")
        if not os.path.exists(p):
            print("!! 缺 %s" % p); continue
        obs, plan, n = round1_mean(p)
        if obs is None:
            print("!! %s 没有 round>=1 的步" % r); continue
        rows.append({"run": r, "n_round1_steps": n,
                     "observed_round1_pct": "%.1f" % (obs * 100),
                     "planned_round1_pct": "%.2f" % (plan * 100)})
    header = "\n".join([
        "# round-1+ 命中率 —— 由 scripts/lab/gen_hitrate.py 生成，勿手改",
        "# 口径：观测的 server_prefix_hit_rate，round>=1，逐步【非加权均值】",
        "# ⚠ 与 reuse.csv 的 reuse_frac 不是一回事（那个是全部步、按 token 加权），",
        "#   两者不可换算、不可混引。planned_round1_pct 是 trace 常量，不随 cap 变。",
        "run,n_round1_steps,observed_round1_pct,planned_round1_pct"])
    fields = ["run", "n_round1_steps", "observed_round1_pct", "planned_round1_pct"]
    body = header + "\n" + "\n".join(",".join(str(x[k]) for k in fields) for x in rows) + "\n"
    assert isinstance(body, str)
    with open(out_p, "w") as fh:
        fh.write(body)
    print("写入 %s：%d 行" % (out_p, len(rows)))
    for x in rows:
        print("  %-20s round>=1 n=%3d  观测 %5s%%  计划 %s%%" % (
            x["run"], x["n_round1_steps"], x["observed_round1_pct"], x["planned_round1_pct"]))


if __name__ == "__main__":
    main()
#!/usr/bin/env python3
"""轮次间隔：trace 要求的 vs 实测的 —— paper/data/toolwait.csv。

为什么要有：论文摘要里那句「the wait averages 16.85 s, not the sub-second figure
the trace's median suggests」的数字【一度不在任何 CSV 里】（2026-09-24 审计发现）。
按本项目纪律，摘要级的数字必须有出处。

口径（必须写清，否则差一个分母就得到完全不同的均值 —— 我自己就先算错过一次）：
  轮次转换 = 【同会话相邻两轮】。等价于「每个会话除最后一轮之外的每一行」，
  因为某一轮的 tool_wait_after_ms 就是到下一轮的等待。**每个会话的最后一轮
  没有后继，必须排除**：不排除的话分母是 999（全部行）而均值只有 12.65 s，
  排除后分母是 799 而均值 15.82 s。这两个数差 25%，全在于分母。
  （trace 的 tool_wait_after_ms 又只有 662 个正值 —— 把零值也滤掉均值变成 19.10 s，
   那是第三种口径。三种都不是错的，但正文必须指明用的是哪一种。）

  要求值 = trace 的 tool_wait_after_ms（重放管线要等这么久）
  实测值 = run 的 steps.jsonl 里，上一轮 complete_timestamp 到下一轮
           submit_timestamp 的实际间隔（含等 permit 的门时间）

用法: python3 gen_toolwait.py <trace.csv> <results_dir> <out.csv> <run> [<run> ...]
"""
import csv, collections, datetime, json, os, statistics, sys


def pct(v, p):
    s = sorted(v)
    return s[min(len(s) - 1, int(p * len(s)))]


def transitions(trace):
    """返回 [(session_id, round_idx, tool_wait_after_ms)]，只含同会话相邻轮。"""
    rows = list(csv.DictReader(open(trace)))
    mx = collections.defaultdict(int)
    for r in rows:
        mx[r["session_id"]] = max(mx[r["session_id"]], int(r["round_idx"]))
    out = [(r["session_id"], int(r["round_idx"]), float(r["tool_wait_after_ms"]))
           for r in rows if int(r["round_idx"]) < mx[r["session_id"]]]
    assert len(out) == len(rows) - len(mx), \
        "轮次转换数 %d != 行数 %d - 会话数 %d" % (len(out), len(rows), len(mx))
    return out, len(rows), len(mx)


def measured(steps_path):
    """实测间隔：同一会话里，上一轮 complete 到下一轮 submit。"""
    st = [json.loads(l) for l in open(steps_path) if l.strip()]
    by = collections.defaultdict(dict)
    for s in st:
        by[s["session_id"]][s["round_idx"]] = s
    out = []
    for sid, d in by.items():
        for k in sorted(d):
            if k + 1 in d:
                out.append((d[k + 1]["submit_timestamp"] - d[k]["complete_timestamp"]) * 1000.0)
    return out


def main():
    trace, root, out_p = sys.argv[1], sys.argv[2], sys.argv[3]
    runs = sys.argv[4:]
    tr, nrows, nsess = transitions(trace)
    dem = [w for _, _, w in tr]

    rows = []
    for r in runs:
        sp = os.path.join(root, r, "steps.jsonl")
        if not os.path.exists(sp):
            print("!! 缺 %s" % sp)
            continue
        me = measured(sp)
        assert len(me) == len(dem), \
            "%s: 实测转换数 %d != trace 的 %d —— 分母不一致，均值不可比" % (r, len(me), len(dem))
        rows.append({"run": r, "n_transitions": len(me),
                     "demand_med_s": "%.2f" % (statistics.median(dem) / 1000),
                     "demand_mean_s": "%.2f" % (statistics.mean(dem) / 1000),
                     "demand_p90_s": "%.2f" % (pct(dem, 0.9) / 1000),
                     "measured_med_s": "%.2f" % (statistics.median(me) / 1000),
                     "measured_mean_s": "%.2f" % (statistics.mean(me) / 1000),
                     "measured_p90_s": "%.2f" % (pct(me, 0.9) / 1000)})

    header = "\n".join([
        "# trace: %s（%d 行 - %d 会话 = %d 次转换）" % (os.path.basename(trace), nrows, nsess, len(dem)),
        "# 轮次间隔：trace 要求 vs 实测 —— 由 scripts/lab/gen_toolwait.py 生成，勿手改",
        "# 轮次转换 = 同会话相邻两轮 = 每会话除最后一轮外的每一行",
        "# ⚠ 分母不同，均值差很多，正文引用时必须指明口径：",
        "#     全部 999 行（含每会话最后一轮，无后继）  均值 12.65 s",
        "#     799 次轮次转换（本表）                  均值 15.82 s   <- 论文用的",
        "#     662 个非零 tool_wait                     均值 19.10 s",
        "# 论文摘要的 16.85 s 是【实测】列的均值（run 的 steps.jsonl 实测间隔，",
        "# 含等 permit 的门时间），15.82 s 是 trace 要求值。",
        "run,n_transitions,demand_med_s,demand_mean_s,demand_p90_s,measured_med_s,measured_mean_s,measured_p90_s"])
    fields = ["run", "n_transitions", "demand_med_s", "demand_mean_s", "demand_p90_s",
              "measured_med_s", "measured_mean_s", "measured_p90_s"]
    body = header + "\n" + "\n".join(",".join(str(r[k]) for k in fields) for r in rows) + "\n"
    assert isinstance(body, str)
    with open(out_p, "w") as fh:
        fh.write(body)

    print("轮次转换 %d（%d 行 - %d 会话）" % (len(dem), nrows, nsess))
    print("要求值: 中位 %.2f s  均值 %.2f s  p90 %.2f s" % (
        statistics.median(dem) / 1000, statistics.mean(dem) / 1000, pct(dem, 0.9) / 1000))
    for r in rows:
        print("  %-24s 实测 中位 %5s  均值 %5s  p90 %5s" % (
            r["run"], r["measured_med_s"], r["measured_mean_s"], r["measured_p90_s"]))


if __name__ == "__main__":
    main()
